# Secret screen: what never enters the knowledge base, the search index or a mirrored note. name<TAB>regex, case-insensitive.
# A text with any hit is held back, or has the match replaced by "[redacted <name>]"; the value is never printed.
# A bare "pass" is deliberately not a pattern (it matches "pass: no" and "PASS:"); a quoted value after "pass" still matches.
password	\b(password|passwd|pwd)\b\s*[:=]\s*\S{4,}
pass_quoted	\bpass\b\s*[:=]\s*["'][^"'\s]{6,}["']
private_key	-----BEGIN [A-Z ]*PRIVATE KEY-----
aws_key	\bAKIA[0-9A-Z]{16}\b
slack_token	\bxox[abprs]-[0-9A-Za-z-]{10,}
anthropic_key	\bsk-ant-[0-9A-Za-z_-]{10,}
sk_key	\bsk-[0-9A-Za-z]{20,}
github_token	\bgh[pousr]_[0-9A-Za-z]{30,}
bearer	\bbearer\s+[0-9A-Za-z._-]{20,}
key_assign	\b(api[_-]?key|secret|token|access[_-]?key)\b\s*[:=]\s*['"]?[0-9A-Za-z._/+-]{16,}
conn_string	\b(postgres(ql)?|mysql|mongodb(\+srv)?|redis)://[^\s:/@]+:[^\s@/]+@
jwt	\beyJ[0-9A-Za-z_-]{10,}\.[0-9A-Za-z_-]{10,}\.
atlassian_token	\bATATT[0-9A-Za-z_=-]{20,}
